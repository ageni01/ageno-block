## 第一章：C到C++的演进

### 1.1 C与C++的关系

```cpp
/**
 * C和C++的对比
 * 
 * C语言特点：
 * - 面向过程
 * - 手动内存管理
 * - 没有类和对象
 * - 没有命名空间
 * - 没有引用
 * - 没有函数重载
 * 
 * C++特点：
 * - 支持面向对象（类、继承、多态）
 * - 兼容C语言（几乎）
 * - 支持泛型编程（模板）
 * - 更强的类型检查
 * - 更丰富的标准库（STL）
 * - 引用、命名空间、函数重载等新特性
 */

// ============ C风格代码 ============
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// C结构体
struct Point {
    int x;
    int y;
};

// C函数
void initPoint(struct Point* p, int x, int y) {
    p->x = x;
    p->y = y;
}

void printPoint(struct Point* p) {
    printf("Point(%d, %d)\n", p->x, p->y);
}

int main_c() {
    // 动态内存分配
    struct Point* p = (struct Point*)malloc(sizeof(struct Point));
    initPoint(p, 10, 20);
    printPoint(p);
    free(p);
    
    return 0;
}

// ============ C++风格代码 ============
#include <iostream>
#include <string>

using namespace std;

// C++类
class Point {
private:
    int x, y;
    
public:
    // 构造函数
    Point(int x = 0, int y = 0) : x(x), y(y) {}
    
    // 成员函数
    void print() const {
        cout << "Point(" << x << ", " << y << ")" << endl;
    }
    
    // 运算符重载
    Point operator+(const Point& other) const {
        return Point(x + other.x, y + other.y);
    }
};

int main() {
    // 自动管理内存
    Point p1(10, 20);
    Point p2(5, 10);
    
    p1.print();
    
    Point p3 = p1 + p2;  // 运算符重载
    p3.print();
    
    return 0;
}
```

### 2.4 引用（C++特有）

```cpp
#include <iostream>
using namespace std;

/**
 * 引用（Reference）- C++的重要特性
 * 引用是变量的别名，与原变量共享同一块内存
 * 
 * 引用 vs 指针：
 * - 引用必须初始化，指针可以不初始化
 * - 引用不能改变指向，指针可以
 * - 引用没有nullptr，指针有
 * - 引用更安全、更易用
 */
void reference_demo() {
    // ========== 基本引用 ==========
    int x = 10;
    int& ref = x;  // ref是x的引用（别名）
    
    cout << "x = " << x << endl;      // 10
    cout << "ref = " << ref << endl;  // 10
    
    ref = 20;  // 通过引用修改x
    cout << "x = " << x << endl;      // 20
    
    // 引用和原变量的地址相同
    cout << "&x = " << &x << endl;
    cout << "&ref = " << &ref << endl;
    
    // ========== 引用作为函数参数 ==========
    // 值传递（复制）
    void func_by_value(int x);
    
    // 引用传递（不复制，可修改原变量）
    void func_by_reference(int& x);
    
    // 常量引用（不复制，不可修改）
    void func_by_const_ref(const int& x);
    
    int a = 100;
    func_by_value(a);      // a不变
    func_by_reference(a);  // a可能被修改
    func_by_const_ref(a);  // a不会被修改
    
    // ========== 引用作为返回值 ==========
    // 返回引用可以作为左值
    int& get_element(int arr[], int index) {
        return arr[index];
    }
    
    int arr[5] = {1, 2, 3, 4, 5};
    get_element(arr, 2) = 100;  // 修改arr[2]
    
    // ========== 右值引用（C++11）==========
    int&& rref = 10;  // 右值引用，绑定到临时对象
    
    // 移动语义
    string s1 = "Hello";
    string s2 = std::move(s1);  // s1的内容移动到s2，s1变为空
}

// 交换两个变量（使用引用）
void swap(int& a, int& b) {
    int temp = a;
    a = b;
    b = temp;
}

// 常量引用作为参数（效率高，安全）
void print(const string& str) {
    cout << str << endl;
    // str = "modify";  // 编译错误：不能修改常量引用
}
```

```cpp
/**
 * 运算符重载
 * 让自定义类型可以使用运算符

class Complex {
private:
    double real, imag;
    
public:
    Complex(double r = 0, double i = 0) : real(r), imag(i) {}
    
    // 成员函数重载
    Complex operator+(const Complex& other) const {
        return Complex(real + other.real, imag + other.imag);
    }
    
    Complex operator-(const Complex& other) const {
        return Complex(real - other.real, imag - other.imag);
    }
    
    Complex operator*(const Complex& other) const {
        return Complex(
            real * other.real - imag * other.imag,
            real * other.imag + imag * other.real
        );
    }
    
    // 自增自减
    Complex& operator++() {  // 前置++
        ++real;
        return *this;
    }
    
    Complex operator++(int) {  // 后置++
        Complex temp = *this;
        ++real;
        return temp;
    }
    
    // 赋值运算符
    Complex& operator=(const Complex& other) {
        if (this != &other) {
            real = other.real;
            imag = other.imag;
        }
        return *this;
    }
    
    // 复合赋值运算符
    Complex& operator+=(const Complex& other) {
        real += other.real;
        imag += other.imag;
        return *this;
    }
    
    // 比较运算符
    bool operator==(const Complex& other) const {
        return real == other.real && imag == other.imag;
    }
    
    bool operator!=(const Complex& other) const {
        return !(*this == other);
    }
    
    // 下标运算符
    double& operator[](int index) {
        return index == 0 ? real : imag;
    }
    
    // 函数调用运算符
    double operator()() const {
        return sqrt(real * real + imag * imag);  // 模
    }
    
    // 类型转换运算符
    operator double() const {
        return real;
    }
    
    // 友元函数重载（全局函数）
    friend ostream& operator<<(ostream& os, const Complex& c);
    friend istream& operator>>(istream& is, Complex& c);
    
    // 友元函数：支持 5 + c（左操作数不是Complex）
    friend Complex operator+(double d, const Complex& c);
};
```

```cpp
/**
 * C++11: Lambda表达式
 * 匿名函数，函数式编程
 * 
 * 语法：[捕获列表](参数列表) -> 返回类型 { 函数体 }
 */

void lambda_demo() {
    // 基本lambda
    auto hello = []() {
        cout << "Hello, Lambda!" << endl;
    };
    hello();
    
    // 带参数
    auto add = [](int a, int b) {
        return a + b;
    };
    cout << "add(3, 5) = " << add(3, 5) << endl;
    
    // 指定返回类型
    auto divide = [](double a, double b) -> double {
        if (b == 0) return 0;
        return a / b;
    };
    
    // ========== 捕获列表 ==========
    int x = 10, y = 20;
    
    // [] 不捕获任何变量
    auto lambda1 = []() { 
        // cout << x;  // 错误：不能访问x
    };
    
    // [=] 值捕获（拷贝）
    auto lambda2 = [=]() {
        cout << "x = " << x << ", y = " << y << endl;
        // x = 100;  // 错误：不能修改
    };
    
    // [&] 引用捕获（可修改）
    auto lambda3 = [&]() {
        x = 100;
        y = 200;
    };
    lambda3();
    cout << "After lambda3: x = " << x << ", y = " << y << endl;
    
    // [x] 只捕获x（值）
    auto lambda4 = [x]() {
        cout << "x = " << x << endl;
    };
    
    // [&x] 只捕获x（引用）
    auto lambda5 = [&x]() {
        x = 300;
    };
    
    // [=, &x] 默认值捕获，x引用捕获
    auto lambda6 = [=, &x]() {
        x = 400;
        cout << "y = " << y << endl;
    };
    
    // [this] 捕获this指针（在类中）
    // [*this] 捕获this对象的拷贝（C++17）
    
    // mutable：允许修改值捕获的变量
    int z = 10;
    auto lambda7 = [z]() mutable {
        z = 100;  // 只修改副本，不影响外部z
        cout << "Inside lambda: z = " << z << endl;
    };
    lambda7();
    cout << "Outside lambda: z = " << z << endl;  // 10
    
    // ========== Lambda的应用 ==========
    
    // 1. STL算法
    vector<int> vec = {5, 2, 8, 1, 9, 3};
    
    // 排序
    sort(vec.begin(), vec.end(), [](int a, int b) {
        return a > b;  // 降序
    });
    
    // 查找
    auto it = find_if(vec.begin(), vec.end(), [](int x) {
        return x > 5;
    });
    
    // 计数
    int count = count_if(vec.begin(), vec.end(), [](int x) {
        return x % 2 == 0;
```

```cpp
class String {
private:
    char* data;
    size_t length;
    
public:
    // 构造函数
    String(const char* str = "") {
        length = strlen(str);
        data = new char[length + 1];
        strcpy(data, str);
        cout << "Constructor: " << data << endl;
    }
    
    // 拷贝构造函数（深拷贝）
    String(const String& other) {
        length = other.length;
        data = new char[length + 1];
        strcpy(data, other.data);
        cout << "Copy constructor: " << data << endl;
    }
    
    // 移动构造函数（转移所有权）
    String(String&& other) noexcept {
        data = other.data;
        length = other.length;
        other.data = nullptr;
        other.length = 0;
        cout << "Move constructor" << endl;
    }
    
    // 拷贝赋值运算符
    String& operator=(const String& other) {
        if (this != &other) {
            delete[] data;
            length = other.length;
            data = new char[length + 1];
            strcpy(data, other.data);
            cout << "Copy assignment" << endl;
        }
        return *this;
    }
    
    // 移动赋值运算符
    String& operator=(String&& other) noexcept {
        if (this != &other) {
            delete[] data;
            data = other.data;
            length = other.length;
            other.data = nullptr;
            other.length = 0;
            cout << "Move assignment" << endl;
        }
        return *this;
    }
    
    ~String() {
        delete[] data;
    }
    
    void print() const {
        if (data) cout << data << endl;
    }
};
```

```cpp
    struct promise_type {

                std::coroutine_handle<promise_type>::from_promise(*this)
            };
        }

        return !handle.done();
    }

    }
```

```cpp
    // lower_bound：第一个>=的元素
    auto lb = lower_bound(vec.begin(), vec.end(), 5);
    
    // upper_bound：第一个>的元素
    auto ub = upper_bound(vec.begin(), vec.end(), 5);
    
    // ========== 排序 ==========
    
    // sort：排序
    sort(vec.begin(), vec.end());  // 升序
    sort(vec.begin(), vec.end(), greater<int>());  // 降序
    
    // 自定义比较器
    sort(vec.begin(), vec.end(), [](int a, int b) {
        return abs(a) < abs(b);
    });
    
    // stable_sort：稳定排序
    stable_sort(vec.begin(), vec.end());
    
    // partial_sort：部分排序
    partial_sort(vec.begin(), vec.begin() + 3, vec.end());  // 前3个最小
    
    // nth_element：第n个元素
    nth_element(vec.begin(), vec.begin() + 4, vec.end());  // 第5小的元素
    
    // ========== 修改 ==========
    
    // fill：填充
    fill(vec.begin(), vec.end(), 0);
    
    // fill_n：填充n个
    fill_n(vec.begin(), 5, 1);
    
    // generate：生成
    generate(vec.begin(), vec.end(), []() { return rand() % 100; });
    
    // transform：转换
    vector<int> squared(vec.size());
    transform(vec.begin(), vec.end(), squared.begin(), [](int x) {
        return x * x;
    });
    
    // replace：替换
    replace(vec.begin(), vec.end(), 0, 10);  // 0替换为10
    
    // replace_if：条件替换
    replace_if(vec.begin(), vec.end(), [](int x) {
        return x < 5;
    }, 0);
    
    // remove：移除（逻辑删除）
    auto new_end = remove(vec.begin(), vec.end(), 0);
    vec.erase(new_end, vec.end());  // 物理删除
    
    // remove_if：条件移除
    vec.erase(
        remove_if(vec.begin(), vec.end(), [](int x) {
            return x % 2 == 0;
        }),
        vec.end()
    );
    
    // unique：去重（需要已排序）
    sort(vec.begin(), vec.end());
    auto last = unique(vec.begin(), vec.end());
    vec.erase(last, vec.end());
    
    // reverse：反转
    reverse(vec.begin(), vec.end());
    
    // rotate：旋转
    rotate(vec.begin(), vec.begin() + 3, vec.end());
    
    // shuffle：打乱（C++11）
    random_device rd;
    mt19937 g(rd());
    shuffle(vec.begin(), vec.end(), g);
    
    // ========== 复制 ==========
    
    // copy：复制
    vector<int> dest(vec.size());
    copy(vec.begin(), vec.end(), dest.begin());
    
    // copy_if：条件复制
    vector<int> evens;
    copy_if(vec.begin(), vec.end(), back_inserter(evens), [](int x) {
        return x % 2 == 0;
    });
    
    // copy_n：复制n个
    copy_n(vec.begin(), 5, dest.begin());
    
    // ========== 数值算法 ==========
    
    // accumulate：求和
    int sum = accumulate(vec.begin(), vec.end(), 0);
    
    // 自定义操作
    int product = accumulate(vec.begin(), vec.end(), 1, multiplies<int>());
    
    // partial_sum：部分和
    vector<int> partial(vec.size());
    partial_sum(vec.begin(), vec.end(), partial.begin());
    
    // adjacent_difference：相邻差
    vector<int> diff(vec.size());
    adjacent_difference(vec.begin(), vec.end(), diff.begin());
    
    // inner_product：内积
    int dot = inner_product(vec.begin(), vec.end(), vec.begin(), 0);
    
    // ========== 集合操作（需要已排序）==========
    
    vector<int> v1{1, 2, 3, 4, 5};
    vector<int> v2{3, 4, 5, 6, 7};
    vector<int> result;
    
    // 并集
    set_union(v1.begin(), v1.end(), v2.begin(), v2.end(), 
              back_inserter(result));
    
    // 交集
    result.clear();
    set_intersection(v1.begin(), v1.end(), v2.begin(), v2.end(),
                     back_inserter(result));
    
    // 差集
    result.clear();
    set_difference(v1.begin(), v1.end(), v2.begin(), v2.end(),
                   back_inserter(result));
    
    // ========== 最值 ==========
    
    // min_element / max_element
    auto min_it = min_element(vec.begin(), vec.end());
    auto max_it = max_element(vec.begin(), vec.end());
    
    // minmax_element（C++11）：同时找最小和最大
    auto [min_it2, max_it2] = minmax_element(vec.begin(), vec.end());
    
    // ========== 排列组合 ==========
    
    // next_permutation：下一个排列
    vector<int> perm{1, 2, 3};
    do {
        for (int x : perm) cout << x << " ";
        cout << endl;
    } while (next_permutation(perm.begin(), perm.end()));
    
    // prev_permutation：上一个排列
    
    // ========== 堆操作 ==========
    
    // make_heap：建堆
    make_heap(vec.begin(), vec.end());
    
    // push_heap：插入
    vec.push_back(100);
    push_heap(vec.begin(), vec.end());
    
    // pop_heap：删除堆顶
    pop_heap(vec.begin(), vec.end());
    vec.pop_back();
    
    // sort_heap：堆排序
    sort_heap(vec.begin(), vec.end());
```

```cpp
void increment() {
    // 1. lock/unlock
    mtx.lock();
    ++shared_data;
    mtx.unlock();
    
    // 2. lock_guard：RAII风格（推荐）
    {
        std::lock_guard<std::mutex> lock(mtx);
        ++shared_data;
    }  // 自动解锁
    
    // 3. unique_lock：更灵活
    {
        std::unique_lock<std::mutex> lock(mtx);
        ++shared_data;
        lock.unlock();  // 可以手动解锁
        // 做其他事情
        lock.lock();    // 可以再次加锁
    }
    
    // 4. try_lock：尝试加锁
    if (mtx.try_lock()) {
        ++shared_data;
        mtx.unlock();
    }
}

 * 递归锁

std::recursive_mutex rec_mtx;

void recursive_function(int count) {
    std::lock_guard<std::recursive_mutex> lock(rec_mtx);

    }

 * 读写锁（C++17）

    std::cout << "Data: " << rw_data << std::endl;
```

### 11.3 条件变量

```cpp
#include <condition_variable>
#include <queue>

/**
 * 条件变量（Condition Variable）
 * 线程间通信
 */

std::mutex cv_mtx;
std::condition_variable cv;
std::queue<int> data_queue;
bool finished = false;

// 生产者
void producer() {
    for (int i = 0; i < 10; ++i) {
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
        
        {
            std::lock_guard<std::mutex> lock(cv_mtx);
            data_queue.push(i);
            std::cout << "Produced: " << i << std::endl;
        }
        
        cv.notify_one();  // 通知一个等待的线程
    }
    
    {
        std::lock_guard<std::mutex> lock(cv_mtx);
        finished = true;
    }
    cv.notify_all();  // 通知所有等待的线程
}

// 消费者
void consumer() {
    while (true) {
        std::unique_lock<std::mutex> lock(cv_mtx);
        
        // 等待条件满足
        cv.wait(lock, []{ return !data_queue.empty() || finished; });
        
        if (!data_queue.empty()) {
            int value = data_queue.front();
            data_queue.pop();
            lock.unlock();  // 提前解锁，提高并发
            
            std::cout << "Consumed: " << value << std::endl;
        } else if (finished) {
            break;
        }
    }
}

void condition_variable_demo() {
    std::thread t1(producer);
    std::thread t2(consumer);
    std::thread t3(consumer);
    
    t1.join();
    t2.join();
    t3.join();
}
```

```cpp
/**
 * 无锁队列示例
 */
template<typename T>
class LockFreeQueue {
private:
    struct Node {
        T data;
        std::atomic<Node*> next;
        Node(T const& data_) : data(data_), next(nullptr) {}
    };
    
    std::atomic<Node*> head;
    std::atomic<Node*> tail;
    
public:
    LockFreeQueue() : head(new Node(T())), tail(head.load()) {}
    
    void push(T const& data) {
        Node* new_node = new Node(data);
        Node* old_tail = tail.load();
        
        while (!tail.compare_exchange_weak(old_tail, new_node)) {
            // 重试
        }
        
        old_tail->next.store(new_node);
    }
    
    bool pop(T& result) {
        Node* old_head = head.load();
        Node* next = old_head->next.load();
        
        if (next == nullptr) {
            return false;  // 队列为空
        }
        
        result = next->data;
        head.store(next);
        delete old_head;
        return true;
    }
};
```

### 12.1 RAII与资源管理

```cpp
/**
 * RAII（Resource Acquisition Is Initialization）
 * 资源获取即初始化
 */

// 文件句柄RAII包装
class FileHandle {
private:
    FILE* file;
    
public:
    FileHandle(const char* filename, const char* mode) 
        : file(fopen(filename, mode)) {
        if (!file) {
            throw std::runtime_error("Failed to open file");
        }
    }
    
    ~FileHandle() {
        if (file) {
            fclose(file);
        }
    }
    
    // 禁止拷贝
    FileHandle(const FileHandle&) = delete;
    FileHandle& operator=(const FileHandle&) = delete;
    
    // 允许移动
    FileHandle(FileHandle&& other) noexcept : file(other.file) {
        other.file = nullptr;
    }
    
    FileHandle& operator=(FileHandle&& other) noexcept {
        if (this != &other) {
            if (file) fclose(file);
            file = other.file;
            other.file = nullptr;
        }
        return *this;
    }
    
    FILE* get() { return file; }
};

// 锁的RAII包装
class Lock {
private:
    std::mutex& mtx;
    
public:
    Lock(std::mutex& m) : mtx(m) {
        mtx.lock();
    }
    
    ~Lock() {
        mtx.unlock();
    }
};

// 通用资源管理器
template<typename Resource, typename Deleter>
class ResourceGuard {
private:
    Resource resource;
    Deleter deleter;
    bool engaged;
    
public:
    ResourceGuard(Resource r, Deleter d) 
        : resource(r), deleter(d), engaged(true) {}
    
    ~ResourceGuard() {
        if (engaged) {
            deleter(resource);
        }
    }
    
    ResourceGuard(const ResourceGuard&) = delete;
    ResourceGuard& operator=(const ResourceGuard&) = delete;
    
    ResourceGuard(ResourceGuard&& other) noexcept
        : resource(other.resource)
        , deleter(std::move(other.deleter))
        , engaged(other.engaged) {
        other.engaged = false;
    }
    
    Resource get() const { return resource; }
    void release() { engaged = false; }
};
```

```cpp
#include <string>
#include <thread>
#include <vector>

class SimpleHTTPServer {
private:
    int server_fd;
    int port;
    
public:
    SimpleHTTPServer(int port) : port(port), server_fd(-1) {}
    
    ~SimpleHTTPServer() {
        if (server_fd >= 0) {
            close(server_fd);
        }
    }
    
    bool start() {
        // 创建socket
        server_fd = socket(AF_INET, SOCK_STREAM, 0);
        if (server_fd < 0) {
            std::cerr << "Failed to create socket" << std::endl;
            return false;
        }
        
        // 设置socket选项
        int opt = 1;
        setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
        
        // 绑定地址
        sockaddr_in address;
        address.sin_family = AF_INET;
        address.sin_addr.s_addr = INADDR_ANY;
        address.sin_port = htons(port);
        
        if (bind(server_fd, (sockaddr*)&address, sizeof(address)) < 0) {
            std::cerr << "Bind failed" << std::endl;
            return false;
        }
        
        // 监听
        if (listen(server_fd, 10) < 0) {
            std::cerr << "Listen failed" << std::endl;
            return false;
        }
        
        std::cout << "Server started on port " << port << std::endl;
        return true;
    }
    
    void run() {
        while (true) {
            sockaddr_in client_addr;
            socklen_t client_len = sizeof(client_addr);
            
            int client_fd = accept(server_fd, (sockaddr*)&client_addr, &client_len);
            if (client_fd < 0) {
                continue;
            }
            
            // 创建新线程处理请求
            std::thread(&SimpleHTTPServer::handleClient, this, client_fd).detach();
        }
    }
    
private:
    void handleClient(int client_fd) {
        char buffer[4096] = {0};
        read(client_fd, buffer, sizeof(buffer));
        
        // 解析HTTP请求
        std::string request(buffer);
        std::istringstream iss(request);
        std::string method, path, protocol;
        iss >> method >> path >> protocol;
        
        std::cout << method << " " << path << std::endl;
        
        // 生成HTTP响应
        std::string response = generateResponse(path);
        write(client_fd, response.c_str(), response.length());
        
        close(client_fd);
    }
    
    std::string generateResponse(const std::string& path) {
        std::string body = R"(
            <!DOCTYPE html>
            <html>
            <head><title>Simple HTTP Server</title></head>
            <body>
                <h1>Welcome to Simple HTTP Server</h1>
                <p>Path: )" + path + R"(</p>
            </body>
            </html>
        )";
        
        std::ostringstream oss;
        oss << "HTTP/1.1 200 OK\r\n";
        oss << "Content-Type: text/html\r\n";
        oss << "Content-Length: " << body.length() << "\r\n";
        oss << "\r\n";
        oss << body;
        
        return oss.str();
    }
};
```

### 13.2 线程池实现

```cpp
/**
 * 线程池实现
 * 演示并发编程、任务队列、RAII等
 */

#include <thread>
#include <mutex>
#include <condition_variable>
#include <queue>
#include <functional>
#include <future>

class ThreadPool {
private:
    std::vector<std::thread> workers;
    std::queue<std::function<void()>> tasks;
    
    std::mutex queue_mutex;
    std::condition_variable condition;
    bool stop;
    
public:
    ThreadPool(size_t num_threads) : stop(false) {
        for (size_t i = 0; i < num_threads; ++i) {
            workers.emplace_back([this] {
                while (true) {
                    std::function<void()> task;
                    
                    {
                        std::unique_lock<std::mutex> lock(queue_mutex);
                        condition.wait(lock, [this] {
                            return stop || !tasks.empty();
                        });
                        
                        if (stop && tasks.empty()) {
                            return;
                        }
                        
                        task = std::move(tasks.front());
                        tasks.pop();
                    }
                    
                    task();
                }
            });
        }
    }
    
    ~ThreadPool() {
        {
            std::unique_lock<std::mutex> lock(queue_mutex);
            stop = true;
        }
        
        condition.notify_all();
        
        for (std::thread& worker : workers) {
            worker.join();
        }
    }
    
    template<typename F, typename... Args>
    auto enqueue(F&& f, Args&&... args) 
        -> std::future<typename std::result_of<F(Args...)>::type> {
        
        using return_type = typename std::result_of<F(Args...)>::type;
        
        auto task = std::make_shared<std::packaged_task<return_type()>>(
            std::bind(std::forward<F>(f), std::forward<Args>(args)...)
        );
        
        std::future<return_type> res = task->get_future();
        
        {
            std::unique_lock<std::mutex> lock(queue_mutex);
            
            if (stop) {
                throw std::runtime_error("enqueue on stopped ThreadPool");
            }
            
            tasks.emplace([task]() { (*task)(); });
        }
        
        condition.notify_one();
        return res;
    }
};

// 使用示例
void thread_pool_demo() {
    ThreadPool pool(4);
    
    std::vector<std::future<int>> results;
    
    for (int i = 0; i < 8; ++i) {
        results.emplace_back(
            pool.enqueue([i] {
                std::cout << "Task " << i << " running on thread " 
                          << std::this_thread::get_id() << std::endl;
                std::this_thread::sleep_for(std::chrono::seconds(1));
                return i * i;
            })
        );
    }
    
    for (auto& result : results) {
        std::cout << "Result: " << result.get() << std::endl;
    }
}
```

```cpp
/**
 * 简化版unique_ptr实现
 * 演示RAII、移动语义、模板编程等
 */

template<typename T>
class MyUniquePtr {
private:
    T* ptr;
    
public:
    // 构造函数
    explicit MyUniquePtr(T* p = nullptr) : ptr(p) {}
    
    // 析构函数
    ~MyUniquePtr() {
        delete ptr;
    }
    
    // 禁止拷贝
    MyUniquePtr(const MyUniquePtr&) = delete;
    MyUniquePtr& operator=(const MyUniquePtr&) = delete;
    
    // 移动构造
    MyUniquePtr(MyUniquePtr&& other) noexcept : ptr(other.ptr) {
        other.ptr = nullptr;
    }
    
    // 移动赋值
    MyUniquePtr& operator=(MyUniquePtr&& other) noexcept {
        if (this != &other) {
            delete ptr;
            ptr = other.ptr;
            other.ptr = nullptr;
        }
        return *this;
    }
    
    // 解引用
    T& operator*() const { return *ptr; }
    T* operator->() const { return ptr; }
    
    // 获取原始指针
    T* get() const { return ptr; }
    
    // 释放所有权
    T* release() {
        T* old_ptr = ptr;
        ptr = nullptr;
        return old_ptr;
    }
    
    // 重置指针
    void reset(T* p = nullptr) {
        delete ptr;
        ptr = p;
    }
    
    // bool转换
    explicit operator bool() const {
        return ptr != nullptr;
    }
};

// 工厂函数
template<typename T, typename... Args>
MyUniquePtr<T> make_my_unique(Args&&... args) {
    return MyUniquePtr<T>(new T(std::forward<Args>(args)...));
}
```

## 常见问题FAQ

### Q1: C++和C有什么区别？
A: C++是C的超集，增加了面向对象、泛型编程、异常处理等特性，但保持了C的高效性。

### Q2: 应该学习哪个C++标准？
A: 建议从C++11开始学习，这是现代C++的分水岭，然后逐步了解C++14/17/20的新特性。

### Q3: 智能指针应该什么时候使用？
A: 几乎总是使用智能指针管理动态内存，默认使用`unique_ptr`，需要共享所有权时使用`shared_ptr`。

### Q4: 什么时候使用引用，什么时候使用指针？
A: 优先使用引用。只有在需要可选参数、需要改变指向对象、需要多态容器时才使用指针。

### Q5: 如何避免内存泄漏？
A: 使用RAII原则，使用智能指针，使用容器管理动态内存，避免手动new/delete。

### Q6: C++的性能优势在哪里？
A: 零开销抽象、编译时计算、内存控制、无垃圾回收开销、可以直接操作硬件。

### Q7: 如何提高C++编程水平？
A: 多读优秀代码、多写项目、参与开源、学习设计模式、关注新标准、参与社区讨论。

```cpp
/**
 * Socket编程基础
 * TCP/IP网络通信
 */

#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <unistd.h>
#include <string.h>
#include <iostream>

/**
 * TCP服务器
 */
class TCPServer {
private:
    int server_fd;
    int port;
    
public:
    TCPServer(int port) : port(port), server_fd(-1) {}
    
    ~TCPServer() {
        if (server_fd >= 0) {
            close(server_fd);
        }
    }
    
    bool start() {
        // 1. 创建socket
        server_fd = socket(AF_INET, SOCK_STREAM, 0);
        if (server_fd < 0) {
            std::cerr << "socket() failed" << std::endl;
            return false;
        }
        
        // 2. 设置socket选项（地址重用）
        int opt = 1;
        if (setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt)) < 0) {
            std::cerr << "setsockopt() failed" << std::endl;
            return false;
        }
        
        // 3. 绑定地址和端口
        sockaddr_in address;
        memset(&address, 0, sizeof(address));
        address.sin_family = AF_INET;
        address.sin_addr.s_addr = INADDR_ANY;  // 监听所有网卡
        address.sin_port = htons(port);
        
        if (bind(server_fd, (sockaddr*)&address, sizeof(address)) < 0) {
            std::cerr << "bind() failed" << std::endl;
            return false;
        }
        
        // 4. 监听
        if (listen(server_fd, SOMAXCONN) < 0) {
            std::cerr << "listen() failed" << std::endl;
            return false;
        }
        
        std::cout << "Server listening on port " << port << std::endl;
        return true;
    }
    
    void run() {
        while (true) {
            sockaddr_in client_addr;
            socklen_t client_len = sizeof(client_addr);
            
            // 5. 接受连接
            int client_fd = accept(server_fd, (sockaddr*)&client_addr, &client_len);
            if (client_fd < 0) {
                std::cerr << "accept() failed" << std::endl;
                continue;
            }
            
            // 获取客户端信息
            char client_ip[INET_ADDRSTRLEN];
            inet_ntop(AF_INET, &client_addr.sin_addr, client_ip, sizeof(client_ip));
            int client_port = ntohs(client_addr.sin_port);
            std::cout << "Client connected: " << client_ip << ":" << client_port << std::endl;
            
            // 处理客户端（单线程版本）
            handleClient(client_fd);
        }
    }
    
private:
    void handleClient(int client_fd) {
        char buffer[4096];
        
        while (true) {
            // 6. 接收数据
            ssize_t n = recv(client_fd, buffer, sizeof(buffer) - 1, 0);
            if (n <= 0) {
                if (n == 0) {
                    std::cout << "Client disconnected" << std::endl;
                } else {
                    std::cerr << "recv() failed" << std::endl;
                }
                break;
            }
            
            buffer[n] = '\0';
            std::cout << "Received: " << buffer << std::endl;
            
            // 7. 发送数据（回显）
            if (send(client_fd, buffer, n, 0) < 0) {
                std::cerr << "send() failed" << std::endl;
                break;
            }
        }
        
        // 8. 关闭连接
        close(client_fd);
    }
};

/**
 * TCP客户端
 */
class TCPClient {
private:
    int sock_fd;
    std::string server_ip;
    int server_port;
    
public:
    TCPClient(const std::string& ip, int port) 
        : sock_fd(-1), server_ip(ip), server_port(port) {}
    
    ~TCPClient() {
        if (sock_fd >= 0) {
            close(sock_fd);
        }
    }
    
    bool connect() {
        // 1. 创建socket
        sock_fd = socket(AF_INET, SOCK_STREAM, 0);
        if (sock_fd < 0) {
            std::cerr << "socket() failed" << std::endl;
            return false;
        }
        
        // 2. 连接服务器
        sockaddr_in server_addr;
        memset(&server_addr, 0, sizeof(server_addr));
        server_addr.sin_family = AF_INET;
        server_addr.sin_port = htons(server_port);
        
        if (inet_pton(AF_INET, server_ip.c_str(), &server_addr.sin_addr) <= 0) {
            std::cerr << "Invalid address" << std::endl;
            return false;
        }
        
        if (::connect(sock_fd, (sockaddr*)&server_addr, sizeof(server_addr)) < 0) {
            std::cerr << "connect() failed" << std::endl;
            return false;
        }
        
        std::cout << "Connected to " << server_ip << ":" << server_port << std::endl;
        return true;
    }
    
    bool sendData(const std::string& data) {
        ssize_t n = send(sock_fd, data.c_str(), data.length(), 0);
        return n == (ssize_t)data.length();
    }
    
    std::string receiveData() {
        char buffer[4096];
        ssize_t n = recv(sock_fd, buffer, sizeof(buffer) - 1, 0);
        if (n > 0) {
            buffer[n] = '\0';
            return std::string(buffer);
        }
        return "";
    }
};

/**
 * UDP服务器
 */
class UDPServer {
private:
    int sock_fd;
    int port;
    
public:
    UDPServer(int port) : port(port), sock_fd(-1) {}
    
    ~UDPServer() {
        if (sock_fd >= 0) {
            close(sock_fd);
        }
    }
    
    bool start() {
        // UDP使用SOCK_DGRAM
        sock_fd = socket(AF_INET, SOCK_DGRAM, 0);
        if (sock_fd < 0) {
            return false;
        }
        
        sockaddr_in address;
        memset(&address, 0, sizeof(address));
        address.sin_family = AF_INET;
        address.sin_addr.s_addr = INADDR_ANY;
        address.sin_port = htons(port);
        
        if (bind(sock_fd, (sockaddr*)&address, sizeof(address)) < 0) {
            return false;
        }
        
        std::cout << "UDP Server listening on port " << port << std::endl;
        return true;
    }
    
    void run() {
        char buffer[4096];
        sockaddr_in client_addr;
        socklen_t client_len = sizeof(client_addr);
        
        while (true) {
            // UDP使用recvfrom
            ssize_t n = recvfrom(sock_fd, buffer, sizeof(buffer) - 1, 0,

                // UDP使用sendto

                      (sockaddr*)&client_addr, client_len);
```

```cpp
/**

 * 可以同时监听多个文件描述符

#include <sys/select.h>
#include <vector>
#include <algorithm>

class SelectServer {
private:
    int server_fd;
    int port;
    std::vector<int> client_fds;
    
public:
    SelectServer(int port) : port(port), server_fd(-1) {}
    
    bool start() {
        server_fd = socket(AF_INET, SOCK_STREAM, 0);
        if (server_fd < 0) return false;
        
        int opt = 1;
        setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
        
        sockaddr_in address;
        memset(&address, 0, sizeof(address));
        address.sin_family = AF_INET;
        address.sin_addr.s_addr = INADDR_ANY;
        address.sin_port = htons(port);
        
        if (bind(server_fd, (sockaddr*)&address, sizeof(address)) < 0) {
            return false;
        }
        
        if (listen(server_fd, SOMAXCONN) < 0) {
            return false;
        }
        
        std::cout << "Select server listening on port " << port << std::endl;
        return true;
    }
    
    void run() {
        while (true) {
            fd_set read_fds;
            FD_ZERO(&read_fds);
            
            // 添加监听socket
            FD_SET(server_fd, &read_fds);
            int max_fd = server_fd;
            
            // 添加客户端socket
            for (int fd : client_fds) {
                FD_SET(fd, &read_fds);
                max_fd = std::max(max_fd, fd);
            }
            
            // select等待事件
            // 参数：最大fd+1, 读fd集合, 写fd集合, 异常fd集合, 超时时间
            int ret = select(max_fd + 1, &read_fds, nullptr, nullptr, nullptr);
            
            if (ret < 0) {
                std::cerr << "select() failed" << std::endl;
                break;
            }
            
            // 检查监听socket（新连接）
            if (FD_ISSET(server_fd, &read_fds)) {
                sockaddr_in client_addr;
                socklen_t client_len = sizeof(client_addr);
                int client_fd = accept(server_fd, (sockaddr*)&client_addr, &client_len);
                
                if (client_fd >= 0) {
                    client_fds.push_back(client_fd);
                    std::cout << "New client connected, fd=" << client_fd << std::endl;
                }
            }
            
            // 检查客户端socket（数据到达）
            for (auto it = client_fds.begin(); it != client_fds.end(); ) {
                int fd = *it;
                
                if (FD_ISSET(fd, &read_fds)) {
                    char buffer[4096];
                    ssize_t n = recv(fd, buffer, sizeof(buffer), 0);
                    
                    if (n <= 0) {
                        // 连接关闭或错误
                        std::cout << "Client disconnected, fd=" << fd << std::endl;
                        close(fd);
                        it = client_fds.erase(it);
                    } else {
                        buffer[n] = '\0';
                        std::cout << "Received from fd=" << fd << ": " << buffer << std::endl;
                        
                        // 回显
                        send(fd, buffer, n, 0);
                        ++it;
                    }
                } else {
                    ++it;
                }
            }
        }
    }
};
```

```cpp
class PollServer {
private:
    int server_fd;

public:

            return false;
        }
        
        if (listen(server_fd, SOMAXCONN) < 0) {
            return false;
        }
        
        // 添加监听socket到poll数组
        pollfd pfd;
        pfd.fd = server_fd;
        pfd.events = POLLIN;  // 监听可读事件
        poll_fds.push_back(pfd);

        while (true) {
            // poll等待事件
            // 参数：pollfd数组, 数组大小, 超时时间(-1表示永久等待)
            int ret = poll(poll_fds.data(), poll_fds.size(), -1);
            
            if (ret < 0) {
                std::cerr << "poll() failed" << std::endl;
                break;
            }
            
            // 遍历所有fd
            for (size_t i = 0; i < poll_fds.size(); ++i) {
                if (poll_fds[i].revents & POLLIN) {
                    if (poll_fds[i].fd == server_fd) {
                        // 监听socket有新连接
                        sockaddr_in client_addr;
                        socklen_t client_len = sizeof(client_addr);
                        int client_fd = accept(server_fd, (sockaddr*)&client_addr, &client_len);
                        
                        if (client_fd >= 0) {
                            pollfd pfd;
                            pfd.fd = client_fd;
                            pfd.events = POLLIN;
                            poll_fds.push_back(pfd);
                            std::cout << "New client connected, fd=" << client_fd << std::endl;
                        }
                    } else {
                        // 客户端socket有数据
                        int fd = poll_fds[i].fd;
                        char buffer[4096];
                        ssize_t n = recv(fd, buffer, sizeof(buffer), 0);
                        
                        if (n <= 0) {
                            // 连接关闭
                            std::cout << "Client disconnected, fd=" << fd << std::endl;
                            close(fd);
                            poll_fds.erase(poll_fds.begin() + i);
                            --i;  // 调整索引
                        } else {
                            buffer[n] = '\0';
                            std::cout << "Received: " << buffer << std::endl;
                            send(fd, buffer, n, 0);
                        }
                    }
                }
            }
        }
    }
};
```

### 14.4 IO多路复用：epoll（Linux）

```cpp
/**
 * epoll - Linux高性能IO多路复用
 * 只在Linux平台可用
 */

#include <sys/epoll.h>
#include <fcntl.h>
#include <unordered_map>

/**
 * 设置非阻塞模式
 */
bool setNonBlocking(int fd) {
    int flags = fcntl(fd, F_GETFL, 0);
    if (flags < 0) return false;
    return fcntl(fd, F_SETFL, flags | O_NONBLOCK) >= 0;
}

class EpollServer {
private:
    int server_fd;
    int epoll_fd;
    int port;
    
    static const int MAX_EVENTS = 1024;
    epoll_event events[MAX_EVENTS];
    
public:
    EpollServer(int port) : port(port), server_fd(-1), epoll_fd(-1) {}
    
    ~EpollServer() {
        if (epoll_fd >= 0) close(epoll_fd);
        if (server_fd >= 0) close(server_fd);
    }
    
    bool start() {
        // 1. 创建监听socket
        server_fd = socket(AF_INET, SOCK_STREAM, 0);
        if (server_fd < 0) return false;
        
        int opt = 1;
        setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
        
        // 设置非阻塞
        setNonBlocking(server_fd);
        
        sockaddr_in address;
        memset(&address, 0, sizeof(address));
        address.sin_family = AF_INET;
        address.sin_addr.s_addr = INADDR_ANY;
        address.sin_port = htons(port);
        
        if (bind(server_fd, (sockaddr*)&address, sizeof(address)) < 0) {
            return false;
        }
        
        if (listen(server_fd, SOMAXCONN) < 0) {
            return false;
        }
        
        // 2. 创建epoll实例
        epoll_fd = epoll_create1(0);
        if (epoll_fd < 0) {
            return false;
        }
        
        // 3. 添加监听socket到epoll
        epoll_event ev;
        ev.events = EPOLLIN | EPOLLET;  // 边缘触发模式
        ev.data.fd = server_fd;
        
        if (epoll_ctl(epoll_fd, EPOLL_CTL_ADD, server_fd, &ev) < 0) {
            return false;
        }
        
        std::cout << "Epoll server listening on port " << port << std::endl;
        return true;
    }
    
    void run() {
        while (true) {
            // 4. 等待事件
            // 参数：epoll实例, 事件数组, 最大事件数, 超时时间
            int n = epoll_wait(epoll_fd, events, MAX_EVENTS, -1);
            
            if (n < 0) {
                std::cerr << "epoll_wait() failed" << std::endl;
                break;
            }
            
            // 5. 处理事件
            for (int i = 0; i < n; ++i) {
                if (events[i].data.fd == server_fd) {
                    // 新连接
                    handleNewConnection();
                } else {
                    // 客户端数据
                    handleClientData(events[i].data.fd);
                }
            }
        }
    }
    
private:
    void handleNewConnection() {
        while (true) {
            sockaddr_in client_addr;
            socklen_t client_len = sizeof(client_addr);
            int client_fd = accept(server_fd, (sockaddr*)&client_addr, &client_len);
            
            if (client_fd < 0) {
                if (errno == EAGAIN || errno == EWOULDBLOCK) {
                    // 所有连接都已处理
                    break;
                }
                std::cerr << "accept() failed" << std::endl;
                break;
            }
            
            // 设置非阻塞
            setNonBlocking(client_fd);
            
            // 添加到epoll
            epoll_event ev;
            ev.events = EPOLLIN | EPOLLET;  // 边缘触发
            ev.data.fd = client_fd;
            
            if (epoll_ctl(epoll_fd, EPOLL_CTL_ADD, client_fd, &ev) < 0) {
                close(client_fd);
                continue;
            }
            
            char client_ip[INET_ADDRSTRLEN];
            inet_ntop(AF_INET, &client_addr.sin_addr, client_ip, sizeof(client_ip));
            std::cout << "New client: " << client_ip << ", fd=" << client_fd << std::endl;
        }
    }
    
    void handleClientData(int fd) {
        char buffer[4096];
        
        while (true) {
            ssize_t n = recv(fd, buffer, sizeof(buffer), 0);
            
            if (n > 0) {
                buffer[n] = '\0';
                std::cout << "Received from fd=" << fd << ": " << buffer << std::endl;
                
                // 回显
                send(fd, buffer, n, 0);
            } else if (n == 0) {
                // 连接关闭
                std::cout << "Client disconnected, fd=" << fd << std::endl;
                epoll_ctl(epoll_fd, EPOLL_CTL_DEL, fd, nullptr);
                close(fd);
                break;
            } else {
                if (errno == EAGAIN || errno == EWOULDBLOCK) {
                    // 数据读取完毕
                    break;
                }
                // 错误
                std::cerr << "recv() failed" << std::endl;
                epoll_ctl(epoll_fd, EPOLL_CTL_DEL, fd, nullptr);
                close(fd);
                break;
            }
        }
    }
};

/**
 * epoll的优势：
 * 1. 没有fd数量限制
 * 2. 性能不随fd数量增加而降低（O(1)）
 * 3. 只返回就绪的fd，不需要遍历
 * 4. 支持边缘触发（ET）和水平触发（LT）
 * 
 * epoll触发模式：
 * LT（水平触发）：默认模式，只要有数据就通知
 * ET（边缘触发）：只在状态变化时通知一次，需要一次性读完所有数据
 * 
 * epoll操作：
 * EPOLL_CTL_ADD：添加fd
 * EPOLL_CTL_MOD：修改fd事件
 * EPOLL_CTL_DEL：删除fd
 * 
 * epoll事件：
 * EPOLLIN：可读
 * EPOLLOUT：可写
 * EPOLLET：边缘触发
 * EPOLLONESHOT：只触发一次
 */
```

```cpp
class EventLoop {
private:
    int epoll_fd;
    bool running;
    std::unordered_map<int, std::shared_ptr<Connection>> connections;
    
public:
    EventLoop() : running(false) {
        epoll_fd = epoll_create1(0);
    }

        if (epoll_fd >= 0) {
            close(epoll_fd);
        }
    }

        int fd = conn->getFd();
        connections[fd] = conn;
        
        epoll_event ev;
        ev.events = EPOLLIN | EPOLLET;
        ev.data.fd = fd;
        epoll_ctl(epoll_fd, EPOLL_CTL_ADD, fd, &ev);
    }
    
    void removeConnection(int fd) {
        epoll_ctl(epoll_fd, EPOLL_CTL_DEL, fd, nullptr);
        connections.erase(fd);
    }
    
    void loop() {
        running = true;
        epoll_event events[1024];
        
        while (running) {
            int n = epoll_wait(epoll_fd, events, 1024, -1);
            
            for (int i = 0; i < n; ++i) {
                int fd = events[i].data.fd;
                auto it = connections.find(fd);
                
                if (it != connections.end()) {
                    auto conn = it->second;
                    
                    if (events[i].events & EPOLLIN) {
                        conn->handleRead();
                    }
                    
                    // 可以添加EPOLLOUT、EPOLLERR等处理
                }
            }
        }
    }
    
    void stop() {
        running = false;
    }
};
```

### 15.1 线程池进阶

```cpp
/**
 * 高级线程池实现
 * 支持任务优先级、动态调整线程数、任务统计等
 */

#include <thread>
#include <mutex>
#include <condition_variable>
#include <queue>
#include <functional>
#include <future>
#include <atomic>
#include <chrono>

class AdvancedThreadPool {
private:
    // 任务结构
    struct Task {
        std::function<void()> func;
        int priority;
        
        bool operator<(const Task& other) const {
            return priority < other.priority;  // 优先队列大顶堆
        }
    };
    
    std::vector<std::thread> workers;
    std::priority_queue<Task> tasks;
    
    mutable std::mutex queue_mutex;
    std::condition_variable condition;
    std::atomic<bool> stop;
    std::atomic<int> idle_threads;
    std::atomic<int> total_tasks;
    
    // 动态调整参数
    size_t min_threads;
    size_t max_threads;
    std::chrono::seconds thread_timeout;
    
public:
    AdvancedThreadPool(size_t min_threads = 4, 
                      size_t max_threads = 16,
                      std::chrono::seconds timeout = std::chrono::seconds(60))
        : stop(false)
        , idle_threads(0)
        , total_tasks(0)
        , min_threads(min_threads)
        , max_threads(max_threads)
        , thread_timeout(timeout) {
        
        // 创建最小线程数
        for (size_t i = 0; i < min_threads; ++i) {
            addWorker();
        }
    }
    
    ~AdvancedThreadPool() {
        stop = true;
        condition.notify_all();
        
        for (std::thread& worker : workers) {
            if (worker.joinable()) {
                worker.join();
            }
        }
    }
    
    // 提交任务（带优先级）
    template<typename F, typename... Args>
    auto submit(int priority, F&& f, Args&&... args) 
        -> std::future<typename std::result_of<F(Args...)>::type> {
        
        using return_type = typename std::result_of<F(Args...)>::type;
        
        auto task = std::make_shared<std::packaged_task<return_type()>>(
            std::bind(std::forward<F>(f), std::forward<Args>(args)...)
        );
        
        std::future<return_type> res = task->get_future();
        
        {
            std::unique_lock<std::mutex> lock(queue_mutex);
            
            if (stop) {
                throw std::runtime_error("ThreadPool is stopped");
            }
            
            tasks.push(Task{[task]() { (*task)(); }, priority});
            ++total_tasks;
        }
        
        // 动态调整线程数
        adjustThreads();
        
        condition.notify_one();
        return res;
    }
    
    // 获取统计信息
    size_t getThreadCount() const {
        return workers.size();
    }
    
    size_t getIdleThreadCount() const {
        return idle_threads.load();
    }
    
    size_t getTaskCount() const {
        std::lock_guard<std::mutex> lock(queue_mutex);
        return tasks.size();
    }
    
    size_t getTotalTasksProcessed() const {
        return total_tasks.load();
    }
    
private:
    void addWorker() {
        workers.emplace_back([this] {
            while (true) {
                Task task;
                
                {
                    std::unique_lock<std::mutex> lock(queue_mutex);
                    
                    ++idle_threads;
                    
                    // 等待任务或超时
                    bool timeout = !condition.wait_for(lock, thread_timeout, [this] {
                        return stop || !tasks.empty();
                    });
                    
                    --idle_threads;
                    
                    // 超时且线程数超过最小值，退出线程
                    if (timeout && workers.size() > min_threads) {
                        return;
                    }
                    
                    if (stop && tasks.empty()) {
                        return;
                    }
                    
                    if (!tasks.empty()) {
                        task = std::move(const_cast<Task&>(tasks.top()));
                        tasks.pop();
                    }
                }
                
                if (task.func) {
                    task.func();
                }
            }
        });
    }
    
    void adjustThreads() {
        size_t current_threads = workers.size();
        size_t pending_tasks = tasks.size();
        size_t idle = idle_threads.load();
        
        // 如果任务多且空闲线程少，增加线程
        if (pending_tasks > idle && current_threads < max_threads) {
            std::lock_guard<std::mutex> lock(queue_mutex);
            if (workers.size() < max_threads) {
                addWorker();
            }
        }
    }
};

// 使用示例
void advanced_threadpool_demo() {
    AdvancedThreadPool pool(2, 8);
    
    std::vector<std::future<int>> results;
    
    // 提交不同优先级的任务
    for (int i = 0; i < 10; ++i) {
        int priority = i % 3;  // 优先级0-2
        results.push_back(pool.submit(priority, [i] {
            std::cout << "Task " << i << " running\n";
            std::this_thread::sleep_for(std::chrono::milliseconds(100));
            return i * i;
        }));
    }
    
    // 输出统计信息
    std::cout << "Thread count: " << pool.getThreadCount() << "\n";
    std::cout << "Idle threads: " << pool.getIdleThreadCount() << "\n";
    std::cout << "Pending tasks: " << pool.getTaskCount() << "\n";
    
    // 等待结果
    for (auto& result : results) {
        std::cout << "Result: " << result.get() << "\n";
    }
    
    std::cout << "Total tasks processed: " << pool.getTotalTasksProcessed() << "\n";
}
```

```cpp
/**
 * 无锁数据结构
 * 使用CAS（Compare-And-Swap）实现
 */

#include <atomic>

/**
 * 无锁栈
 */
template<typename T>
class LockFreeStack {
private:
    struct Node {
        T data;
        Node* next;
        Node(const T& data) : data(data), next(nullptr) {}
    };
    
    std::atomic<Node*> head;
    
public:
    LockFreeStack() : head(nullptr) {}
    
    ~LockFreeStack() {
        while (Node* node = head.load()) {
            head.store(node->next);
            delete node;
        }
    }
    
    void push(const T& data) {
        Node* new_node = new Node(data);
        new_node->next = head.load(std::memory_order_relaxed);
        
        // CAS循环
        while (!head.compare_exchange_weak(new_node->next, new_node,
                                          std::memory_order_release,
                                          std::memory_order_relaxed)) {
            // 如果失败，new_node->next已被更新为当前head值
            // 继续尝试
        }
    }
    
    bool pop(T& result) {
        Node* old_head = head.load(std::memory_order_relaxed);
        
        while (old_head && 
               !head.compare_exchange_weak(old_head, old_head->next,
                                          std::memory_order_acquire,
                                          std::memory_order_relaxed)) {
            // 重试
        }
        
        if (old_head) {
            result = old_head->data;
            delete old_head;
            return true;
        }
        
        return false;
    }
};

/**
 * 无锁队列（MPMC - Multiple Producer Multiple Consumer）
 */
template<typename T>
class LockFreeQueue {
private:
    struct Node {
        T data;
        std::atomic<Node*> next;
        Node() : next(nullptr) {}
        Node(const T& data) : data(data), next(nullptr) {}
    };
    
    std::atomic<Node*> head;
    std::atomic<Node*> tail;
    
public:
    LockFreeQueue() {
        Node* dummy = new Node();
        head.store(dummy);
        tail.store(dummy);
    }
    
    ~LockFreeQueue() {
        while (Node* node = head.load()) {
            head.store(node->next);
            delete node;
        }
    }
    
    void enqueue(const T& data) {
        Node* new_node = new Node(data);
        
        while (true) {
            Node* last = tail.load();
            Node* next = last->next.load();
            
            if (last == tail.load()) {
                if (next == nullptr) {
                    if (last->next.compare_exchange_weak(next, new_node)) {
                        tail.compare_exchange_weak(last, new_node);
                        return;
                    }
                } else {
                    tail.compare_exchange_weak(last, next);
                }
            }
        }
    }
    
    bool dequeue(T& result) {
        while (true) {
            Node* first = head.load();
            Node* last = tail.load();
            Node* next = first->next.load();
            
            if (first == head.load()) {
                if (first == last) {
                    if (next == nullptr) {
                        return false;  // 队列为空
                    }
                    tail.compare_exchange_weak(last, next);
                } else {
                    result = next->data;
                    if (head.compare_exchange_weak(first, next)) {
                        delete first;
                        return true;
                    }
                }
            }
        }
    }
};

/**
 * ABA问题和解决方案
 */
template<typename T>
class ABAFreeLockFreeStack {
private:
    struct Node {
        T data;
        Node* next;
        Node(const T& data) : data(data), next(nullptr) {}
    };
    
    struct TaggedPointer {
        Node* ptr;
        uintptr_t tag;
    };
    
    std::atomic<TaggedPointer> head;
    
public:
    ABAFreeLockFreeStack() {
        head.store({nullptr, 0});
    }
    
    void push(const T& data) {
        Node* new_node = new Node(data);
        TaggedPointer old_head = head.load();
        
        do {
            new_node->next = old_head.ptr;
        } while (!head.compare_exchange_weak(
            old_head, 
            {new_node, old_head.tag + 1}));  // 增加tag解决ABA问题
    }
    
    bool pop(T& result) {
        TaggedPointer old_head = head.load();
        
        do {
            if (!old_head.ptr) {
                return false;
            }
        } while (!head.compare_exchange_weak(
            old_head,
            {old_head.ptr->next, old_head.tag + 1}));
        
        result = old_head.ptr->data;
        delete old_head.ptr;
        return true;
    }
};
```

```cpp
/**

#include <atomic>
#include <thread>

/**
 * 内存顺序类型：
 * 
 * 1. memory_order_relaxed：最宽松，只保证原子性
 * 2. memory_order_consume：读操作，依赖顺序（已弃用）
 * 3. memory_order_acquire：读操作，获取语义
 * 4. memory_order_release：写操作，释放语义
 * 5. memory_order_acq_rel：读写操作，获取-释放语义
 * 6. memory_order_seq_cst：顺序一致性（默认，最严格）
 */

// 示例1：自旋锁
class SpinLock {
private:
    std::atomic_flag flag = ATOMIC_FLAG_INIT;
    
public:
    void lock() {
        while (flag.test_and_set(std::memory_order_acquire)) {
            // 自旋等待
            std::this_thread::yield();
        }
    }
    
    void unlock() {
        flag.clear(std::memory_order_release);
    }
};

// 示例2：生产者-消费者（使用acquire-release）
class ProducerConsumer {
private:
    std::atomic<int> data{0};
    std::atomic<bool> ready{false};
    
public:
    // 生产者
    void produce(int value) {
        data.store(value, std::memory_order_relaxed);
        ready.store(true, std::memory_order_release);  // 释放语义
    }
    
    // 消费者
    int consume() {
        while (!ready.load(std::memory_order_acquire)) {  // 获取语义
            std::this_thread::yield();
        }
        return data.load(std::memory_order_relaxed);
    }
};

// 示例3：双检锁单例（使用acquire-release）
class Singleton {
private:
    static std::atomic<Singleton*> instance;
    static std::mutex mtx;
    
    Singleton() = default;
    
public:
    static Singleton* getInstance() {
        Singleton* tmp = instance.load(std::memory_order_acquire);
        if (tmp == nullptr) {
            std::lock_guard<std::mutex> lock(mtx);
            tmp = instance.load(std::memory_order_relaxed);
            if (tmp == nullptr) {
                tmp = new Singleton();
                instance.store(tmp, std::memory_order_release);
            }
        }
        return tmp;
    }
};
```

### 15.4 读写锁实现

```cpp
/**
 * 读写锁实现
 * 允许多个读者或一个写者
 */

class ReadWriteLock {
private:
    std::mutex mtx;
    std::condition_variable read_cv, write_cv;
    int readers{0};
    int writers{0};
    int write_waiters{0};
    
public:
    // 读锁
    void readLock() {
        std::unique_lock<std::mutex> lock(mtx);
        // 如果有写者或等待的写者，等待
        while (writers > 0 || write_waiters > 0) {
            read_cv.wait(lock);
        }
        ++readers;
    }
    
    void readUnlock() {
        std::unique_lock<std::mutex> lock(mtx);
        --readers;
        if (readers == 0) {
            write_cv.notify_one();  // 唤醒一个写者
        }
    }
    
    // 写锁
    void writeLock() {
        std::unique_lock<std::mutex> lock(mtx);
        ++write_waiters;
        // 等待所有读者和写者完成
        while (readers > 0 || writers > 0) {
            write_cv.wait(lock);
        }
        --write_waiters;
        ++writers;
    }
    
    void writeUnlock() {
        std::unique_lock<std::mutex> lock(mtx);
        --writers;
        // 优先唤醒写者
        if (write_waiters > 0) {
            write_cv.notify_one();
        } else {
            read_cv.notify_all();  // 唤醒所有读者
        }
    }
};

// RAII封装
class ReadLockGuard {
private:
    ReadWriteLock& lock;
    
public:
    ReadLockGuard(ReadWriteLock& l) : lock(l) {
        lock.readLock();
    }
    
    ~ReadLockGuard() {
        lock.readUnlock();
    }
};

class WriteLockGuard {
private:
    ReadWriteLock& lock;
    
public:
    WriteLockGuard(ReadWriteLock& l) : lock(l) {
        lock.writeLock();
    }
    
    ~WriteLockGuard() {
        lock.writeUnlock();
    }
};

// 使用示例
void rwlock_demo() {

    std::vector<int> shared_data;

        for (int i = 0; i < 10; ++i) {

            std::this_thread::sleep_for(std::chrono::milliseconds(100));
        }
    };
    
    // 写线程

            std::this_thread::sleep_for(std::chrono::milliseconds(200));
        }
    };
    
    std::thread t1(reader);
    std::thread t2(reader);
    std::thread t3(writer);
    
    t1.join();
    t2.join();
    t3.join();
}
```
