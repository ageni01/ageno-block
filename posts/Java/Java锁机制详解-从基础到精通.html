1. 锁的基础概念

1.1 什么是锁

锁（Lock）：
用于控制多个线程对共享资源的访问，保证数据的一致性和完整性。

核心作用：
1. 互斥性（Mutual Exclusion）：同一时刻只有一个线程能访问
2. 可见性（Visibility）：一个线程的修改对其他线程可见
3. 有序性（Ordering）：禁止指令重排序

1.2 锁的分类

按功能分类：
├── 互斥锁（Mutex）：同一时刻只有一个线程能持有
├── 读写锁（Read-Write Lock）：读读共享，读写互斥，写写互斥
├── 自旋锁（Spin Lock）：线程忙等待，不放弃CPU
├── 递归锁（Reentrant Lock）：同一线程可多次获取
└── 公平锁/非公平锁：按申请顺序获取 vs 竞争获取

按实现分类：
├── 悲观锁：假设会发生冲突，先加锁
│   └── synchronized、ReentrantLock
└── 乐观锁：假设不会冲突，使用时检测
    └── CAS（Compare-And-Swap）

按粒度分类：
├── 对象锁：锁住对象实例
├── 类锁：锁住Class对象
└── 分段锁：锁住数据的一部分（ConcurrentHashMap）

按状态分类（JVM层面）：
├── 无锁
├── 偏向锁（Biased Lock）
├── 轻量级锁（Lightweight Lock）
└── 重量级锁（Heavyweight Lock）

2. 悲观锁与乐观锁

2.1 悲观锁（Pessimistic Lock）

概念与特点

/**
 * 悲观锁：假设一定会发生冲突，每次访问数据都加锁
 * 
 * 特点：
 * 1. 独占锁：其他线程阻塞等待
 * 2. 适合写多读少的场景
 * 3. 性能开销较大
 */
public class PessimisticLockDemo {
    private int count = 0;
    
    // synchronized：悲观锁的典型实现
    public synchronized void increment() {
        count++;  // 独占访问，其他线程阻塞
    }
    
    // ReentrantLock：另一种悲观锁实现
    private final ReentrantLock lock = new ReentrantLock();
    
    public void decrement() {
        lock.lock();
        try {
            count--;
        } finally {
            lock.unlock();
        }
    }
}

2.2 乐观锁（Optimistic Lock）

概念与特点

/**
 * 乐观锁：假设不会发生冲突，使用时检测冲突
 * 
 * 特点：
 * 1. 无锁机制：不阻塞其他线程
 * 2. 适合读多写少的场景
 * 3. 性能较好，但可能需要重试
 */
public class OptimisticLockDemo {
    private AtomicInteger count = new AtomicInteger(0);
    
    // CAS：乐观锁的核心实现
    public void increment() {
        int oldValue, newValue;
        do {
            oldValue = count.get();           // 读取当前值
            newValue = oldValue + 1;          // 计算新值
        } while (!count.compareAndSet(oldValue, newValue));  // CAS更新
        
        // 如果更新失败（被其他线程修改），重试
    }
    
    // AtomicInteger内部使用CAS实现
    public void incrementSimple() {
        count.incrementAndGet();  // 等价于上面的循环
    }
}

CAS原理详解

/**
 * CAS（Compare-And-Swap）：比较并交换
 * 
 * 原子操作：
 * boolean compareAndSet(expectedValue, newValue) {
 *     if (currentValue == expectedValue) {
 *         currentValue = newValue;
 *         return true;
 *     }
 *     return false;
 * }
 */
public class CASDemo {
    public static void main(String[] args) {
        AtomicInteger atomic = new AtomicInteger(0);
        
        // 线程1
        new Thread(() -> {
            int expect = 0;
            int update = 1;
            boolean success = atomic.compareAndSet(expect, update);
            System.out.println("Thread1: " + success);  // true
        }).start();
        
        // 线程2
        new Thread(() -> {
            int expect = 0;
            int update = 2;
            boolean success = atomic.compareAndSet(expect, update);
            System.out.println("Thread2: " + success);  // false（已被线程1修改）
        }).start();
    }
}

CAS底层实现

// AtomicInteger源码
public class AtomicInteger {
    private static final Unsafe unsafe = Unsafe.getUnsafe();
    private static final long valueOffset;
    
    static {
        try {
            // 获取value字段的内存偏移量
            valueOffset = unsafe.objectFieldOffset
                (AtomicInteger.class.getDeclaredField("value"));
        } catch (Exception ex) { throw new Error(ex); }
    }
    
    private volatile int value;
    
    public final boolean compareAndSet(int expect, int update) {
        // 调用Unsafe的CAS方法（native方法）
        return unsafe.compareAndSwapInt(this, valueOffset, expect, update);
    }
}

CAS的ABA问题

/**
 * ABA问题：
 * 线程1读取A，准备改为B
 * 线程2将A改为B，又改回A
 * 线程1的CAS成功，但中间状态已改变
 */
public class ABAProblemDemo {
    private static AtomicInteger atomic = new AtomicInteger(100);
    
    public static void main(String[] args) throws InterruptedException {
        // 线程1
        Thread t1 = new Thread(() -> {
            int value = atomic.get();  // 读取100
            System.out.println("T1读取：" + value);
            
            try {
                Thread.sleep(1000);  // 模拟处理时间
            } catch (InterruptedException e) {}
            
            boolean success = atomic.compareAndSet(value, 200);
            System.out.println("T1修改：" + success + ", 当前值：" + atomic.get());
        });
        
        // 线程2：制造ABA
        Thread t2 = new Thread(() -> {
            try {
                Thread.sleep(100);
            } catch (InterruptedException e) {}
            
            atomic.compareAndSet(100, 200);
            System.out.println("T2修改为200");
            atomic.compareAndSet(200, 100);
            System.out.println("T2修改回100");
        });
        
        t1.start();
        t2.start();
        t1.join();
        t2.join();
        
        // 输出：
        // T1读取：100
        // T2修改为200
        // T2修改回100
        // T1修改：true, 当前值：200
        // T1的CAS成功了，但期间值被改过
    }
}

解决ABA问题：AtomicStampedReference

/**
 * AtomicStampedReference：带版本号的CAS
 * 每次修改都更新版本号，避免ABA问题
 */
public class AtomicStampedReferenceDemo {
    // 初始值100，版本号0
    private static AtomicStampedReference<Integer> asr = 
        new AtomicStampedReference<>(100, 0);
    
    public static void main(String[] args) throws InterruptedException {
        // 线程1
        Thread t1 = new Thread(() -> {
            int[] stampHolder = new int[1];
            int value = asr.get(stampHolder);  // 读取值和版本号
            int stamp = stampHolder[0];
            System.out.println("T1读取：value=" + value + ", stamp=" + stamp);
            
            try {
                Thread.sleep(1000);
            } catch (InterruptedException e) {}
            
            // CAS：检查值和版本号
            boolean success = asr.compareAndSet(value, 200, stamp, stamp + 1);
            System.out.println("T1修改：" + success);
        });
        
        // 线程2：制造ABA
        Thread t2 = new Thread(() -> {
            try {
                Thread.sleep(100);
            } catch (InterruptedException e) {}
            
            int[] stampHolder = new int[1];
            int value = asr.get(stampHolder);
            int stamp = stampHolder[0];
            
            asr.compareAndSet(value, 200, stamp, stamp + 1);
            System.out.println("T2修改为200");
            
            value = asr.get(stampHolder);
            stamp = stampHolder[0];
            asr.compareAndSet(value, 100, stamp, stamp + 1);
            System.out.println("T2修改回100");
        });
        
        t1.start();
        t2.start();
        t1.join();
        t2.join();
        
        // 输出：
        // T1读取：value=100, stamp=0
        // T2修改为200
        // T2修改回100
        // T1修改：false  ← 版本号不匹配，修改失败
    }
}

/**

public class ReentrantDemo {

        // method2

/**
 * 特性2：互斥性（Mutual Exclusion）
 * 同一时刻只有一个线程能执行同步代码
 */
public class MutualExclusionDemo {
    private int count = 0;
    
    public synchronized void increment() {
        count++;
    }
    
    public static void main(String[] args) throws InterruptedException {
        MutualExclusionDemo demo = new MutualExclusionDemo();
        
        // 1000个线程，各执行1000次
        Thread[] threads = new Thread[1000];
        for (int i = 0; i < threads.length; i++) {
            threads[i] = new Thread(() -> {
                for (int j = 0; j < 1000; j++) {
                    demo.increment();
                }
            });
            threads[i].start();
        }
        
        for (Thread t : threads) {
            t.join();
        }
        
        System.out.println("Count: " + demo.count);  // 1000000（准确）
    }
}

3.4 synchronized锁的升级

锁升级过程（不可逆）：

无锁 → 偏向锁 → 轻量级锁 → 重量级锁

┌──────────┐
│  无锁状态 │ 初始状态
└─────┬────┘
      │ 第一个线程访问
      ▼
┌──────────┐
│  偏向锁   │ 只有一个线程访问，CAS记录线程ID
└─────┬────┘ 无需加锁/解锁操作
      │ 另一个线程尝试获取
      ▼
┌──────────┐
│ 轻量级锁  │ 多个线程交替访问（无竞争）
└─────┬────┘ CAS操作，自旋等待
      │ 线程竞争激烈
      ▼
┌──────────┐
│ 重量级锁  │ 多个线程竞争
└──────────┘ 操作系统互斥量，线程阻塞

偏向锁详解

/**
 * 偏向锁（Biased Locking）
 * 
 * 核心思想：大多数情况下，锁不仅不存在多线程竞争，
 *          而且总是由同一线程多次获得
 * 
 * 实现：在对象头Mark Word中记录线程ID
 *      后续该线程进入/退出同步块时，无需CAS操作
 */
public class BiasedLockDemo {
    public static void main(String[] args) throws InterruptedException {
        // JVM启动4秒后才启用偏向锁
        Thread.sleep(5000);
        
        Object lock = new Object();
        
        synchronized (lock) {
            // 第一次获取锁：变为偏向锁，记录线程ID
            System.out.println("First lock");
        }
        
        synchronized (lock) {
            // 第二次获取锁：检查线程ID，无需CAS
            System.out.println("Second lock");
        }
    }
}

// 偏向锁的Mark Word（64位）：
// |-------54位-------|--2位--|--1位--|--4位--|--1位--|--2位--|
// |   线程ID        |epoch |unused | age  |biased| 01   |

轻量级锁详解

/**
 * 轻量级锁（Lightweight Locking）
 * 
 * 核心思想：多个线程交替执行同步块，不存在实际竞争
 * 
 * 实现：线程在栈中创建Lock Record，
 *      使用CAS将对象头的Mark Word替换为指向Lock Record的指针
 */
public class LightweightLockDemo {
    private static Object lock = new Object();
    
    public static void main(String[] args) throws InterruptedException {
        // 禁用偏向锁
        // VM参数：-XX:-UseBiasedLocking
        
        Thread t1 = new Thread(() -> {
            synchronized (lock) {
                System.out.println("Thread1 acquired lock");
                try {
                    Thread.sleep(100);
                } catch (InterruptedException e) {}
            }
        });
        
        Thread t2 = new Thread(() -> {
            try {
                Thread.sleep(50);  // 错开执行
            } catch (InterruptedException e) {}
            
            synchronized (lock) {
                System.out.println("Thread2 acquired lock");
            }
        });
        
        t1.start();
        t2.start();
        t1.join();
        t2.join();
        
        // 线程交替执行，使用轻量级锁
    }
}

// 轻量级锁的Mark Word：
// |------------------62位-------------------|--2位--|
// |   指向栈中Lock Record的指针              |  00   |

重量级锁详解

/**
 * 重量级锁（Heavyweight Locking）
 * 
 * 核心思想：线程竞争激烈，使用操作系统互斥量（Mutex）
 * 
 * 实现：未获取锁的线程进入阻塞状态，
 *      需要操作系统进行线程调度（用户态→内核态）
 */
public class HeavyweightLockDemo {
    private static Object lock = new Object();
    
    public static void main(String[] args) {
        // 10个线程同时竞争锁
        for (int i = 0; i < 10; i++) {
            new Thread(() -> {
                synchronized (lock) {
                    System.out.println(Thread.currentThread().getName() + 
                                     " acquired lock");
                    try {
                        Thread.sleep(100);  // 持有锁一段时间
                    } catch (InterruptedException e) {}
                }
            }, "Thread-" + i).start();
        }
        
        // 线程竞争激烈，升级为重量级锁
    }
}

// 重量级锁的Mark Word：
// |------------------62位-------------------|--2位--|
// |   指向Monitor对象的指针                  |  10   |

3.5 Monitor对象

/**
 * Monitor（监视器/管程）
 * 
 * synchronized的底层实现依赖于Monitor对象
 * 每个Java对象都可以关联一个Monitor
 */

// Monitor结构（简化版）：
class Monitor {
    Thread owner;              // 持有锁的线程
    int recursions;            // 重入次数
    Queue<Thread> entrySet;    // 等待获取锁的线程队列
    Queue<Thread> waitSet;     // 调用wait()的线程队列
}

public class MonitorDemo {
    private final Object lock = new Object();
    
    public void method1() {
        synchronized (lock) {
            // 1. 进入：尝试获取Monitor
            //    - 成功：owner = 当前线程
            //    - 失败：进入entrySet等待
            
            System.out.println("Critical section");
            
            // 2. 退出：释放Monitor
            //    - owner = null
            //    - 唤醒entrySet中的一个线程
        }
    }
    
    public void method2() {
        synchronized (lock) {
            try {
                // wait()：释放锁，进入waitSet
                lock.wait();
                
                // 被唤醒后，重新竞争锁
            } catch (InterruptedException e) {}
        }
    }
    
    public void method3() {
        synchronized (lock) {
            // notify()：唤醒waitSet中的一个线程
            lock.notify();
            
            // notifyAll()：唤醒waitSet中的所有线程
            lock.notifyAll();
        }
    }
}

3.6 wait/notify机制

/**
 * wait/notify：线程间通信机制
 * 
 * 必须在synchronized块中调用
 */
public class WaitNotifyDemo {
    private final Object lock = new Object();
    private boolean ready = false;
    
    // 生产者
    public void produce() {
        synchronized (lock) {
            System.out.println("生产数据...");
            ready = true;
            lock.notify();  // 唤醒消费者
        }
    }
    
    // 消费者
    public void consume() {
        synchronized (lock) {
            while (!ready) {  // 使用while而不是if
                try {
                    lock.wait();  // 等待生产者通知
                } catch (InterruptedException e) {}
            }
            System.out.println("消费数据");
        }
    }
    
    public static void main(String[] args) {
        WaitNotifyDemo demo = new WaitNotifyDemo();
        
        // 消费者线程
        new Thread(() -> demo.consume()).start();
        
        // 生产者线程
        new Thread(() -> {
            try {

import java.util.LinkedList;
import java.util.Queue;

public class ProducerConsumerDemo {
    private final Queue<Integer> queue = new LinkedList<>();
    private final int MAX_SIZE = 10;
    private final Object lock = new Object();
    
    // 生产者
    class Producer implements Runnable {
        @Override
        public void run() {
            int value = 0;
            while (true) {
                synchronized (lock) {
                    // 队列满了，等待
                    while (queue.size() == MAX_SIZE) {
                        try {
                            lock.wait();
                        } catch (InterruptedException e) {}
                    }
                    
                    // 生产
                    queue.offer(value);
                    System.out.println("生产：" + value);
                    value++;
                    
                    // 通知消费者
                    lock.notifyAll();
                    
                    try {
                        Thread.sleep(100);
                    } catch (InterruptedException e) {}
                }
            }
        }
    }
    
    // 消费者
    class Consumer implements Runnable {
        @Override
        public void run() {
            while (true) {
                synchronized (lock) {
                    // 队列空了，等待
                    while (queue.isEmpty()) {
                        try {
                            lock.wait();
                        } catch (InterruptedException e) {}
                    }
                    
                    // 消费
                    int value = queue.poll();
                    System.out.println("消费：" + value);
                    
                    // 通知生产者
                    lock.notifyAll();
                    
                    try {
                        Thread.sleep(200);
                    } catch (InterruptedException e) {}
                }
            }
        }
    }
    
    public static void main(String[] args) {
        ProducerConsumerDemo demo = new ProducerConsumerDemo();
        
        // 启动2个生产者
        new Thread(demo.new Producer()).start();
        new Thread(demo.new Producer()).start();
        
        // 启动3个消费者
        new Thread(demo.new Consumer()).start();
        new Thread(demo.new Consumer()).start();
        new Thread(demo.new Consumer()).start();
    }

4.2 ReentrantLock（可重入锁）

import java.util.concurrent.locks.ReentrantLock;

/**
 * ReentrantLock：可重入的互斥锁
 * 
 * 特点：
 * 1. 可重入：同一线程可多次获取
 * 2. 可公平/不公平
 * 3. 可中断
 * 4. 可超时
 */
public class ReentrantLockDemo {
    private final ReentrantLock lock = new ReentrantLock();
    private int count = 0;
    
    // 基本使用
    public void increment() {
        lock.lock();
        try {
            count++;
        } finally {
            lock.unlock();  // 必须在finally中释放
        }
    }
    
    // 可重入性
    public void reentrant() {
        lock.lock();
        try {
            System.out.println("第一次获取锁");
            reentrantInner();  // 可以再次获取
        } finally {
            lock.unlock();
        }
    }
    
    private void reentrantInner() {
        lock.lock();
        try {
            System.out.println("第二次获取锁");
        } finally {
            lock.unlock();
        }
    }
    
    // 尝试获取锁
    public boolean tryIncrement() {
        if (lock.tryLock()) {
            try {
                count++;
                return true;
            } finally {
                lock.unlock();
            }
        }
        return false;  // 获取失败，立即返回
    }
    
    // 超时获取锁
    public boolean incrementWithTimeout() {
        try {
            if (lock.tryLock(1, TimeUnit.SECONDS)) {
                try {
                    count++;
                    return true;
                } finally {
                    lock.unlock();
                }
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
        return false;
    }
    
    // 可中断获取锁
    public void incrementInterruptibly() throws InterruptedException {
        lock.lockInterruptibly();  // 可以被中断
        try {
            count++;
        } finally {
            lock.unlock();
        }
    }
}

多Condition示例：有界队列

import java.util.LinkedList;
import java.util.Queue;
import java.util.concurrent.locks.Condition;
import java.util.concurrent.locks.Lock;
import java.util.concurrent.locks.ReentrantLock;

/**
 * 使用多个Condition实现生产者-消费者
 */
public class BoundedQueue<T> {
    private final Queue<T> queue = new LinkedList<>();
    private final int capacity;
    private final Lock lock = new ReentrantLock();
    private final Condition notFull = lock.newCondition();   // 队列未满
    private final Condition notEmpty = lock.newCondition();  // 队列非空
    
    public BoundedQueue(int capacity) {
        this.capacity = capacity;
    }
    
    // 生产
    public void put(T item) throws InterruptedException {
        lock.lock();
        try {
            while (queue.size() == capacity) {
                notFull.await();  // 等待队列未满
            }
            
            queue.offer(item);
            System.out.println("生产：" + item + ", 队列大小：" + queue.size());
            
            notEmpty.signal();  // 通知消费者
        } finally {
            lock.unlock();
        }
    }
    
    // 消费
    public T take() throws InterruptedException {
        lock.lock();
        try {
            while (queue.isEmpty()) {
                notEmpty.await();  // 等待队列非空
            }
            
            T item = queue.poll();
            System.out.println("消费：" + item + ", 队列大小：" + queue.size());
            
            notFull.signal();  // 通知生产者
            
            return item;
        } finally {
            lock.unlock();
        }
    }
    
    public static void main(String[] args) {
        BoundedQueue<Integer> queue = new BoundedQueue<>(5);
        
        // 生产者
        new Thread(() -> {
            for (int i = 0; i < 10; i++) {
                try {
                    queue.put(i);
                    Thread.sleep(100);
                } catch (InterruptedException e) {}
            }
        }, "Producer").start();
        
        // 消费者
        new Thread(() -> {
            for (int i = 0; i < 10; i++) {
                try {
                    queue.take();
                    Thread.sleep(200);
                } catch (InterruptedException e) {}
            }
        }, "Consumer").start();
    }
}

读写锁的锁降级

import java.util.concurrent.locks.ReadWriteLock;
import java.util.concurrent.locks.ReentrantReadWriteLock;

/**
 * 锁降级：写锁降级为读锁
 * 
 * 允许：持有写锁的同时获取读锁，然后释放写锁
 * 不允许：锁升级（持有读锁时获取写锁）→ 死锁
 */
public class LockDowngradingDemo {
    private final ReadWriteLock rwLock = new ReentrantReadWriteLock();
    private volatile boolean cacheValid = false;
    private Object data;
    
    public void processData() {
        rwLock.readLock().lock();
        if (!cacheValid) {
            // 缓存失效，需要更新
            rwLock.readLock().unlock();  // 释放读锁
            rwLock.writeLock().lock();    // 获取写锁
            
            try {
                // 双重检查
                if (!cacheValid) {
                    data = loadData();     // 加载数据
                    cacheValid = true;
                }
                
                // 锁降级：持有写锁的同时获取读锁
                rwLock.readLock().lock();
            } finally {
                rwLock.writeLock().unlock();  // 释放写锁
            }
        }
        
        try {
            useData(data);  // 使用数据
        } finally {
            rwLock.readLock().unlock();
        }
    }
    
    private Object loadData() {
        System.out.println("加载数据...");
        return new Object();
    }
    
    private void useData(Object data) {
        System.out.println("使用数据：" + data);
    }
}

4.5 StampedLock（JDK 8）

import java.util.concurrent.locks.StampedLock;

/**
 * StampedLock：更快的读写锁
 * 
 * 特点：
 * 1. 乐观读：不加锁，读完验证
 * 2. 悲观读：加读锁
 * 3. 写锁：独占
 * 4. 不可重入
 * 5. 不支持Condition
 * 
 * 性能：比ReentrantReadWriteLock更快
 */
public class StampedLockDemo {
    private final StampedLock sl = new StampedLock();
    private double x, y;
    
    // 写操作
    public void move(double deltaX, double deltaY) {
        long stamp = sl.writeLock();  // 获取写锁
        try {
            x += deltaX;
            y += deltaY;
        } finally {
            sl.unlockWrite(stamp);  // 释放写锁
        }
    }
    
    // 乐观读
    public double distanceFromOrigin() {
        long stamp = sl.tryOptimisticRead();  // 乐观读
        double currentX = x;
        double currentY = y;
        
        if (!sl.validate(stamp)) {  // 验证：期间有写操作？
            stamp = sl.readLock();  // 升级为悲观读锁
            try {
                currentX = x;
                currentY = y;
            } finally {
                sl.unlockRead(stamp);
            }
        }
        
        return Math.sqrt(currentX * currentX + currentY * currentY);
    }
    
    // 悲观读
    public double distanceFromOriginPessimistic() {
        long stamp = sl.readLock();  // 获取读锁
        try {
            return Math.sqrt(x * x + y * y);
        } finally {
            sl.unlockRead(stamp);  // 释放读锁
        }
    }
    
    // 锁转换
    public void moveWithConversion(double deltaX, double deltaY) {
        long stamp = sl.readLock();  // 先获取读锁
        try {
            while (x == 0.0 && y == 0.0) {
                // 尝试转换为写锁
                long ws = sl.tryConvertToWriteLock(stamp);
                if (ws != 0L) {
                    stamp = ws;  // 转换成功
                    x = deltaX;
                    y = deltaY;
                    break;
                } else {
                    // 转换失败，释放读锁，获取写锁
                    sl.unlockRead(stamp);
                    stamp = sl.writeLock();
                }
            }
        } finally {
            sl.unlock(stamp);
        }
    }
}

5. 读写锁深入

5.1 读写锁的特点

/**
 * 读写锁（ReadWriteLock）规则：
 * 
 * 1. 读-读：共享，可以同时进行
 * 2. 读-写：互斥，不能同时进行
 * 3. 写-写：互斥，不能同时进行
 * 4. 写-读：互斥，不能同时进行
 * 
 * ┌─────────┬──────┬──────┬──────┐
 * │         │ 读   │ 写   │      │
 * ├─────────┼──────┼──────┤      │
 * │ 读      │ ✓    │ ✗    │      │
 * │ 写      │ ✗    │ ✗    │      │
 * └─────────┴──────┴──────┴──────┘
 */

import java.util.HashMap;
import java.util.Map;
import java.util.concurrent.locks.ReadWriteLock;
import java.util.concurrent.locks.ReentrantReadWriteLock;

/**
 * 应用：缓存实现
 */
public class CacheWithRWLock<K, V> {
    private final Map<K, V> cache = new HashMap<>();
    private final ReadWriteLock lock = new ReentrantReadWriteLock();
    
    // 读取缓存
    public V get(K key) {
        lock.readLock().lock();
        try {
            System.out.println(Thread.currentThread().getName() + " 读取：" + key);
            return cache.get(key);
        } finally {
            lock.readLock().unlock();
        }
    }
    
    // 写入缓存
    public void put(K key, V value) {
        lock.writeLock().lock();
        try {
            System.out.println(Thread.currentThread().getName() + " 写入：" + key);
            cache.put(key, value);
        } finally {
            lock.writeLock().unlock();
        }
    }
    
    // 清空缓存
    public void clear() {
        lock.writeLock().lock();
        try {
            System.out.println(Thread.currentThread().getName() + " 清空缓存");
            cache.clear();
        } finally {
            lock.writeLock().unlock();
        }
    }
    
    public static void main(String[] args) {
        CacheWithRWLock<String, Integer> cache = new CacheWithRWLock<>();
        
        // 先写入一些数据
        cache.put("key1", 1);
        cache.put("key2", 2);
        
        // 10个读线程
        for (int i = 0; i < 10; i++) {
            new Thread(() -> {
                for (int j = 0; j < 3; j++) {
                    cache.get("key1");
                    try {
                        Thread.sleep(100);
                    } catch (InterruptedException e) {}
                }
            }, "Reader-" + i).start();
        }
        
        // 2个写线程
        for (int i = 0; i < 2; i++) {
            int value = i;
            new Thread(() -> {
                cache.put("key" + value, value * 100);
                try {
                    Thread.sleep(500);
                } catch (InterruptedException e) {}
            }, "Writer-" + i).start();
        }
    }
}

/**

public class LockCoarseningDemo {

        }
        synchronized (lock) {
            // 操作3

    // JIT优化后：合并为一次加锁

            // 操作1

6.3 自旋锁与自适应自旋

/**
 * 自旋锁（Spin Lock）：
 * 线程不放弃CPU，而是执行忙循环（自旋）等待锁
 * 
 * 优势：避免线程切换开销
 * 劣势：占用CPU资源
 * 
 * 适用场景：锁持有时间很短
 */
public class SpinLockDemo {
    private AtomicReference<Thread> owner = new AtomicReference<>();
    
    // 简单自旋锁实现
    public void lock() {
        Thread current = Thread.currentThread();
        
        // 自旋：不断尝试CAS
        while (!owner.compareAndSet(null, current)) {
            // 空循环，忙等待
        }
    }
    
    public void unlock() {
        Thread current = Thread.currentThread();
        owner.compareAndSet(current, null);
    }
    
    public static void main(String[] args) {
        SpinLockDemo lock = new SpinLockDemo();
        
        for (int i = 0; i < 5; i++) {
            new Thread(() -> {
                lock.lock();
                try {
                    System.out.println(Thread.currentThread().getName() + " acquired lock");
                    Thread.sleep(10);  // 短时间持有锁
                } catch (InterruptedException e) {
                } finally {
                    lock.unlock();
                }
            }).start();
        }
    }
}

6.4 锁优化总结

JVM锁优化技术：

1. 偏向锁（Biased Locking）
   └─ 只有一个线程访问，记录线程ID，无需CAS

2. 轻量级锁（Lightweight Locking）
   └─ 多线程交替访问，使用CAS，自旋等待

3. 重量级锁（Heavyweight Locking）
   └─ 竞争激烈，使用操作系统互斥量

4. 自旋锁（Spin Lock）
   └─ 不放弃CPU，忙等待

5. 自适应自旋（Adaptive Spinning）
   └─ 动态调整自旋次数

6. 锁粗化（Lock Coarsening）
   └─ 合并多次加锁/解锁

7. 锁消除（Lock Elimination）
   └─ 消除不必要的锁

配置参数：
-XX:+UseBiasedLocking           # 启用偏向锁（默认）
-XX:BiasedLockingStartupDelay=0 # 偏向锁启动延迟（默认4秒）
-XX:-UseBiasedLocking           # 禁用偏向锁
-XX:+DoEscapeAnalysis           # 启用逃逸分析（默认）
-XX:+EliminateLocks             # 启用锁消除（默认）

7. 死锁问题

7.1 死锁的四个必要条件

死锁（Deadlock）的四个必要条件：

1. 互斥条件（Mutual Exclusion）
   └─ 资源不能被多个线程共享

2. 请求与保持条件（Hold and Wait）
   └─ 持有资源的同时等待其他资源

3. 不剥夺条件（No Preemption）
   └─ 资源不能被强行剥夺

4. 循环等待条件（Circular Wait）
   └─ 存在资源的循环等待链

破坏任意一个条件，即可避免死锁

/**
 * 方法1：按顺序加锁
 * 破坏循环等待条件
 */
public class AvoidDeadlock1 {
    private static final Object lock1 = new Object();
    private static final Object lock2 = new Object();
    
    // 两个线程都按相同顺序加锁
    private static void method1() {
        synchronized (lock1) {  // 先lock1
            synchronized (lock2) {  // 再lock2
                System.out.println("method1");
            }
        }
    }
    
    private static void method2() {
        synchronized (lock1) {  // 先lock1
            synchronized (lock2) {  // 再lock2
                System.out.println("method2");
            }
        }
    }
}

/**
 * 方法2：使用tryLock()超时
 * 破坏请求与保持条件
 */
public class AvoidDeadlock2 {
    private final ReentrantLock lock1 = new ReentrantLock();
    private final ReentrantLock lock2 = new ReentrantLock();
    
    public void transfer() {
        while (true) {
            try {
                if (lock1.tryLock(100, TimeUnit.MILLISECONDS)) {
                    try {
                        if (lock2.tryLock(100, TimeUnit.MILLISECONDS)) {
                            try {
                                // 转账操作
                                System.out.println("转账成功");
                                break;
                            } finally {
                                lock2.unlock();
                            }
                        }
                    } finally {
                        lock1.unlock();
                    }
                }
                
                // 获取锁失败，释放已持有的锁，重试
                Thread.sleep(10);
                
            } catch (InterruptedException e) {
                break;
            }
        }
    }
}

/**
 * 方法3：使用Lock的tryLock()
 * 尝试获取所有锁，失败则全部释放
 */
public class AvoidDeadlock3 {
    private final ReentrantLock lock1 = new ReentrantLock();
    private final ReentrantLock lock2 = new ReentrantLock();
    
    public boolean transfer() {
        // 尝试获取lock1
        if (!lock1.tryLock()) {
            return false;
        }
        
        try {
            // 尝试获取lock2
            if (!lock2.tryLock()) {
                return false;  // 失败，lock1会在finally中释放
            }
            
            try {
                // 转账操作
                System.out.println("转账成功");
                return true;
            } finally {
                lock2.unlock();
            }
        } finally {
            lock1.unlock();
        }
    }
}

 * 方法4：使用哲学家就餐问题的解决方案

 */
public class DiningPhilosophers {
    private final Semaphore semaphore = new Semaphore(4);  // 最多4人同时吃
    private final ReentrantLock[] forks = new ReentrantLock[5];
    
    public DiningPhilosophers() {

            forks[i] = new ReentrantLock();

    }
    
    public void eat(int philosopher) {

            semaphore.acquire();  // 限制同时进餐人数
            
            int left = philosopher;
            int right = (philosopher + 1) % 5;
            
            forks[left].lock();
            forks[right].lock();
            
            try {
                System.out.println("哲学家" + philosopher + "在进餐");
                Thread.sleep(1000);
            } finally {
                forks[right].unlock();
                forks[left].unlock();
            }
            
        } catch (InterruptedException e) {
        } finally {
            semaphore.release();
        }
    }
}

#include <mutex>
#include <condition_variable>
#include <queue>

/**
 * std::condition_variable - 条件变量
 * 对应Java：Condition, wait/notify
 */
class ProducerConsumer {
private:
    std::mutex mtx;
    std::condition_variable cv;
    std::queue<int> queue;
    const size_t MAX_SIZE = 10;

public:
    // 生产者
    void produce(int value) {
        std::unique_lock<std::mutex> lock(mtx);
        
        // 等待队列有空间
        cv.wait(lock, [this]() { return queue.size() < MAX_SIZE; });
        
        queue.push(value);
        std::cout << "生产：" << value << std::endl;
        
        cv.notify_all();  // 通知消费者
    }
    
    // 消费者
    int consume() {
        std::unique_lock<std::mutex> lock(mtx);
        
        // 等待队列非空
        cv.wait(lock, [this]() { return !queue.empty(); });
        
        int value = queue.front();
        queue.pop();
        std::cout << "消费：" << value << std::endl;
        
        cv.notify_all();  // 通知生产者
        
        return value;
    }
};

8.8 性能对比

锁性能对比（从快到慢）：

1. 无锁（Lock-Free）
   └─ std::atomic, CAS
   └─ 最快，但适用场景有限

2. 自旋锁（Spin Lock）
   └─ 适合锁持有时间极短的场景
   └─ 避免线程切换开销

3. 用户态锁（User-Space Lock）
   └─ Java偏向锁、轻量级锁
   └─ C++ std::mutex（通常使用futex）

4. 内核态锁（Kernel-Space Lock）
   └─ Java重量级锁
   └─ pthread_mutex（部分实现）
   └─ 涉及系统调用，最慢

选择建议：
- 读多写少 → 读写锁
- 极短临界区 → 自旋锁
- 一般场景 → 互斥锁
- 无共享状态 → 无锁编程

/**
 * 原则1：缩小锁的范围
 * 只锁必要的代码，减少持锁时间
 */
public class Principle1 {
    private final Object lock = new Object();

    public synchronized void bad() {
        // 不需要同步的操作
        prepareData();
        
        // 需要同步的操作
        updateSharedData();
        
        // 不需要同步的操作
        sendNotification();
    }
    
    // 好：只锁必要的部分
    public void good() {
        prepareData();  // 不加锁
        
        synchronized (lock) {
            updateSharedData();  // 加锁
        }
        
        sendNotification();  // 不加锁
    }
    
    private void prepareData() {}
    private void updateSharedData() {}
    private void sendNotification() {}
}

/**
 * 原则2：减少锁的粒度
 * 使用多个锁分别保护不同的数据
 */
public class Principle2 {

    private final Object lock = new Object();

            data1++;
        }
    }
    
    public void badUpdate2() {
        synchronized (lock) {
            data2++;
        }
    }
    
    // 好：不同的锁保护不同的数据
    private final Object lock1 = new Object();
    private final Object lock2 = new Object();
    
    public void goodUpdate1() {
        synchronized (lock1) {
            data1++;
        }
    }

/**

public class Optimization1 {

    }

/**
 * 技巧2：使用不可变对象
 */
public class Optimization2 {
    // 不好：可变对象需要同步
    private List<String> list = new ArrayList<>();
    
    public synchronized void addBad(String item) {
        list.add(item);
    }
    
    public synchronized List<String> getListBad() {
        return new ArrayList<>(list);  // 防御性拷贝
    }
    
    // 好：不可变对象
    private volatile List<String> immutableList = Collections.emptyList();
    
    public void addGood(String item) {
        List<String> newList = new ArrayList<>(immutableList);
        newList.add(item);
        immutableList = Collections.unmodifiableList(newList);  // 无需同步
    }
    
    public List<String> getListGood() {
        return immutableList;  // 无需拷贝
    }
}

/**
 * 技巧3：延迟初始化
 */
public class Optimization3 {
    // 双重检查锁定（DCL）
    private volatile Helper helper;
    
    public Helper getHelper() {
        if (helper == null) {  // 第一次检查（无锁）
            synchronized (this) {
                if (helper == null) {  // 第二次检查（有锁）
                    helper = new Helper();
                }
            }
        }
        return helper;
    }
    
    // 更好：使用Holder模式（推荐）
    private static class HelperHolder {
        static final Helper INSTANCE = new Helper();
    }
    
    public static Helper getInstance() {
        return HelperHolder.INSTANCE;  // 线程安全，无需锁
    }
    
    private static class Helper {}
}

/**
 * 技巧4：分段锁（Striping）
 */
public class Optimization4<T> {
    private final int STRIPE_COUNT = 16;
    private final Object[] locks = new Object[STRIPE_COUNT];
    private final List<T>[] buckets = new List[STRIPE_COUNT];
    
    public Optimization4() {
        for (int i = 0; i < STRIPE_COUNT; i++) {
            locks[i] = new Object();
            buckets[i] = new ArrayList<>();
        }
    }
    
    public void add(T item) {
        int index = Math.abs(item.hashCode() % STRIPE_COUNT);
        
        synchronized (locks[index]) {  // 只锁一个分段
            buckets[index].add(item);
        }
    }
    
    public boolean contains(T item) {
        int index = Math.abs(item.hashCode() % STRIPE_COUNT);
        
        synchronized (locks[index]) {
            return buckets[index].contains(item);
        }
    }
}

/**
 * 错误2：忘记释放锁
 */
public class Mistake2 {
    private final ReentrantLock lock = new ReentrantLock();
    
    // 错误：异常时不释放锁
    public void bad() {
        lock.lock();
        // 可能抛出异常
        riskyOperation();
        lock.unlock();  // 可能不执行
    }
    
    // 正确：在finally中释放
    public void good() {
        lock.lock();
        try {
            riskyOperation();
        } finally {
            lock.unlock();  // 一定执行
        }
    }
    
    private void riskyOperation() {}
}
